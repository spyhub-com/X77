const express = require("express"); const session = require("express-session"); const passport = require("passport"); const DiscordStrategy = require("passport-discord").Strategy; const axios = require("axios"); const app = express(); const PORT = process.env.PORT || 3000; // ---------------- CONFIGURATION ---------------- const CLIENT_ID = "TON_CLIENT_ID_DISCORD"; const CLIENT_SECRET = "TON_CLIENT_SECRET_DISCORD"; const CALLBACK_URL = "http://localhost:3000/auth/discord/callback"; const BIN_ID = "TON_BIN_ID_JSONBIN"; const API_KEY = "TA_CLE_JSONBIN"; // ---------------- SESSION ---------------- app.use(session({ secret: "axtra-secret", resave: false, saveUninitialized: false })); app.use(passport.initialize()); app.use(passport.session()); passport.serializeUser((user, done) => done(null, user)); passport.deserializeUser((obj, done) => done(null, obj)); passport.use(new DiscordStrategy({ clientID: CLIENT_ID, clientSecret: CLIENT_SECRET, callbackURL: CALLBACK_URL, scope: ["identify"] }, (accessToken, refreshToken, profile, done) => done(null, profile))); // ---------------- AUTH ROUTES ---------------- app.get("/auth/discord", passport.authenticate("discord")); app.get("/auth/discord/callback", passport.authenticate("discord", { failureRedirect: "/" }), (req, res) => res.redirect("/") ); app.get("/auth/discord/logout", (req,res) => { req.logout(() => { res.redirect("/"); }); }); // ---------------- JSONBIN FUNCTIONS ---------------- async function getDB() { try { const res = await axios.get(`https://api.jsonbin.io/v3/b/${BIN_ID}/latest`, { headers: { "X-Master-Key": API_KEY } }); return res.data.record; } catch { return { users: {} }; } } async function saveDB(data) { try { await axios.put(`https://api.jsonbin.io/v3/b/${BIN_ID}`, data, { headers: { "X-Master-Key": API_KEY } }); } catch {} } function canGetCredits(lastReset) { const now = new Date(); const last = new Date(lastReset); return now.getTime() - last.getTime() >= 24*60*60*1000; } // ---------------- FRONT PAGE ---------------- app.get("/", (req, res) => { res.send(` Axtra Credits

Axtra Credits

Connexion avec Discord

Salut !

Crédits actuels :

`); }); // ---------------- API CREDITS ---------------- app.get("/api/credits", async (req,res) => { if(!req.isAuthenticated()) return res.json({ loggedIn:false }); const db = await getDB(); if(!db.users) db.users = {}; let user = db.users[req.user.id]; if(!user) user = { credits:0, lastReset: new Date(0) }; let message = ""; if(canGetCredits(user.lastReset)) { user.credits += 10; user.lastReset = new Date(); db.users[req.user.id] = user; await saveDB(db); message = "✅ Vous avez récupéré vos 10 crédits !"; } else { const remaining = 24*60*60*1000 - (new Date() - new Date(user.lastReset)); const hours = Math.floor(remaining / (1000*60*60)); const minutes = Math.floor((remaining % (1000*60*60)) / (1000*60)); message = \`⏳ Vous devez attendre \${hours}h \${minutes}m avant de récupérer vos crédits.\`; } res.json({ loggedIn:true, username:req.user.username, credits:user.credits, message }); }); // ---------------- LANCEMENT ---------------- app.listen(PORT, () => console.log(\`🚀 Axtra Credits en ligne sur http://localhost:\${PORT}\`));
